--------------------------------------------------------------------------------
  KEY HELPER FUNCTIONS (called from Phase 3 inner loop)
--------------------------------------------------------------------------------

  ending_for_residue(r, m)   ->  (2r^2 + 2r + 1) mod m
                                 fast ull path for m<=1e9, __int128 for k>=10

  reverse_k(val, k)          ->  decimal reverse of k digits

  compute_n_bounds(d)        ->  n_min,n_max  (GMP)   where 2n^2+2n+1 is d-digit

  compute_first_k(n,d,k)     ->  floor(p / 10^(d-k))  (leading k digits of p)

  first_n_with_residue(lo,r,m) ->  smallest n >= lo with n = r (mod m)
  last_n_with_residue (hi,r,m) ->  largest  n <= hi with n = r (mod m)

  sorted_has_value_in_range  ->  binary-search membership in valid_firsts[]

  solve_residues(t, k, out)  ->  Hensel-lift solutions of 2m^2+2m+1 = t (mod 10^k)
                                 base case mod 10, lifts to mod 10^k

  write_checkpoint(maxd,maxk,k,d)  ->  atomic tmp+rename; resumes on restart

--------------------------------------------------------------------------------
  DATA SIZES AT k=10  (mod = 10^10)
--------------------------------------------------------------------------------
   is_valid_ending bitset  : ~1.25 GB   (mod/8 bytes)
   is_valid_first_bs       : ~1.25 GB   (temporary, freed after Phase 2)
   valid_firsts[] sorted   : ~8    GB   (~10.42% of mod, 8 bytes each)
   Total RSS               : ~10   GB   (was ~180 GB before bitset rework)
